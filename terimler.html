## 15. NESNE YONELIMLI PROGRAMLAMA (OOP)[cite: 3, 229, 230]

class Personel:[cite: 245]

    personel_sayisi = 0

    def __init__(self, ad, maas=17000): # Yapici Metot (Constructor)[cite: 248]
        self.ad = ad # Ornek Niteligi (Instance Attribute - Nesneye ozel)[cite: 248]
        self.maas = maas

    def bilgileri_yazdir(self): # Ornek Metodu (Instance Method)[cite: 253]

    def zam_yap(self, artis): # Durum degistiren ornek metodu[cite: 223, 234]
        self.maas += artis[cite: 223, 234]

### Kalitim (Inheritance)[cite: 45]
class Yonetici(Personel):[cite: 45]
    def __init__(self, ad, maas, departman):
        super().__init__(ad, maas) # Ust sinifin yapicisini cagirir
        self.departman = departman

    # Cok Bicimlilik (Polymorphism / Method Overriding)[cite: 45, 46]
    def bilgileri_yazdir(self):[cite: 45]
        print(f"[Yonetici] {self.ad} - Departman: {self.departman} - Maas: {self.maas:,.2f} TL")
